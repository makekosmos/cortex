<script setup lang="ts">
import { computed, onMounted, onUnmounted, shallowRef, watch } from "vue";
import {
  BarChart2,
  FolderSearch,
  Gamepad2,
  PanelLeftClose,
  Settings,
  Star,
} from "lucide-vue-next";
import { RouterLink, useRoute } from "vue-router";
import { useGamesStore } from "../stores/games";
import { useLanguage } from "../composables/useLanguage";
import AppSpotlight from "./AppSpotlight.vue";
import {
  SIDEBAR_MAX_WIDTH,
  SIDEBAR_MIN_WIDTH,
  type SidebarConfig,
} from "../composables/useSidebarConfig";

const props = withDefaults(
  defineProps<{
    mobile?: boolean;
    hidden: boolean;
    config: SidebarConfig;
    showToggle?: boolean;
    showSearch?: boolean;
    reserveTopInset?: boolean;
    enableToggleShortcut?: boolean;
  }>(),
  {
    mobile: false,
    showToggle: true,
    showSearch: true,
    reserveTopInset: true,
    enableToggleShortcut: true,
  },
);

const emit = defineEmits<{
  toggleHidden: [];
  configChange: [config: SidebarConfig];
}>();

const route = useRoute();
const gamesStore = useGamesStore();
const { t } = useLanguage();

const width = shallowRef(props.config.width);
const isResizing = shallowRef(false);
const animating = shallowRef(false);
let animationTimer: number | null = null;
let resizeFrame: number | null = null;

const isMac =
  typeof navigator !== "undefined" && navigator.platform.startsWith("Mac");

watch(
  () => props.config.width,
  (nextWidth) => {
    width.value = nextWidth;
  },
  { immediate: true },
);

const showCollapsedRail = computed(
  () => props.hidden && !props.mobile && props.showToggle,
);

const unmountDesktopContent = computed(
  () => props.hidden && !props.mobile && !props.showToggle,
);

const navItems = [
  { key: "sidebar.library" as const, to: "/", icon: Gamepad2 },
  { key: "sidebar.scan" as const, to: "/scan", icon: FolderSearch },
  { key: "sidebar.statistics" as const, to: "/statistics", icon: BarChart2 },
];

const wrapperStyle = computed(() => {
  if (props.mobile) {
    return undefined;
  }

  if (showCollapsedRail.value) {
    return { width: "44px" };
  }

  if (props.hidden) {
    return { width: "0px" };
  }

  return { width: `${width.value}px` };
});

function isActiveRoute(target: string) {
  return route.path === target || (target === "/" && route.path.startsWith("/game/"));
}

function startAnimation() {
  if (animationTimer !== null) {
    window.clearTimeout(animationTimer);
  }

  animating.value = true;
  animationTimer = window.setTimeout(() => {
    animating.value = false;
    animationTimer = null;
  }, 330);
}

function persistWidth(nextWidth: number) {
  width.value = Math.min(SIDEBAR_MAX_WIDTH, Math.max(SIDEBAR_MIN_WIDTH, nextWidth));
  emit("configChange", {
    ...props.config,
    width: width.value,
  });
}

function toggleHidden() {
  startAnimation();
  emit("toggleHidden");
}

function onMouseMove(event: MouseEvent) {
  if (props.hidden || !isResizing.value) {
    return;
  }

  if (resizeFrame !== null) {
    cancelAnimationFrame(resizeFrame);
  }

  resizeFrame = window.requestAnimationFrame(() => {
    persistWidth(event.clientX);
  });
}

function stopResizing() {
  if (resizeFrame !== null) {
    cancelAnimationFrame(resizeFrame);
    resizeFrame = null;
  }

  isResizing.value = false;
  document.body.classList.remove("sidebar-resizing");
}

function onMouseUp() {
  stopResizing();
}

function handleResizeStart(event: MouseEvent) {
  if (props.mobile) {
    return;
  }

  event.preventDefault();
  isResizing.value = true;
  document.body.classList.add("sidebar-resizing");
}

function onGlobalKeyDown(event: KeyboardEvent) {
  if (!props.enableToggleShortcut || props.mobile) {
    return;
  }

  const rawKey = event.key.toLowerCase();
  const isToggleKey = rawKey === "b" || event.code === "KeyB";
  if (!isToggleKey) {
    return;
  }

  if (!(event.metaKey || event.ctrlKey) || event.altKey || event.shiftKey) {
    return;
  }

  event.preventDefault();
  toggleHidden();
}

onMounted(() => {
  document.addEventListener("keydown", onGlobalKeyDown);
  window.addEventListener("mousemove", onMouseMove);
  window.addEventListener("mouseup", onMouseUp);
});

onUnmounted(() => {
  document.removeEventListener("keydown", onGlobalKeyDown);
  window.removeEventListener("mousemove", onMouseMove);
  window.removeEventListener("mouseup", onMouseUp);
  stopResizing();

  if (animationTimer !== null) {
    window.clearTimeout(animationTimer);
  }
});
</script>

<template>
  <div
    :class="[
      !mobile && 'kosmos-sidebar-wrapper',
      !mobile && showCollapsedRail && 'is-hidden collapsed',
      !mobile && unmountDesktopContent && 'hidden collapsed',
      !mobile && animating && 'animating',
      !mobile && isResizing && 'is-resizing',
      mobile && 'relative flex h-full w-[320px] max-w-[86vw] flex-col',
    ]"
    :style="wrapperStyle"
  >
    <div class="kosmos-sidebar-content">
      <aside
        class="arrancador-sidebar-shell"
        :class="[
          reserveTopInset && !mobile && isMac && 'arrancador-sidebar-shell--mac-safe-top',
          (showToggle || !mobile) && 'arrancador-sidebar-shell--with-top-bar',
        ]"
      >
        <div v-if="showToggle" class="arrancador-sidebar-top">
          <button
            type="button"
            class="arrancador-sidebar-top-toggle"
            :aria-label="hidden ? 'Показать боковую панель' : 'Скрыть боковую панель'"
            :title="hidden ? 'Показать боковую панель' : 'Скрыть боковую панель'"
            @click="toggleHidden"
          >
            <PanelLeftClose class="h-4 w-4" />
          </button>
        </div>

        <template v-if="!showCollapsedRail">
          <div class="arrancador-sidebar-body">
            <template v-if="showSearch">
              <AppSpotlight
                :enable-shortcut="false"
                trigger-variant="sidebar"
                trigger-class-name="arrancador-sidebar-search-button"
              />
              <div class="kosmos-sidebar-divider" />
            </template>

            <RouterLink
              v-for="item in navItems"
              :key="item.to"
              :to="item.to"
              class="kosmos-sidebar-btn"
              :class="{ 'kosmos-sidebar-btn--active': isActiveRoute(item.to) }"
            >
              <component :is="item.icon" class="h-[18px] w-[18px] shrink-0" />
              <span class="truncate">{{ t(item.key) }}</span>
            </RouterLink>

            <template v-if="gamesStore.favorites.length > 0">
              <div class="kosmos-sidebar-divider" />
              <div class="kosmos-sidebar-section-label">{{ t("sidebar.favorites") }}</div>

              <RouterLink
                v-for="game in gamesStore.favorites.slice(0, 5)"
                :key="game.id"
                :to="`/game/${game.id}`"
                class="kosmos-sidebar-project-link"
                :class="{
                  'kosmos-sidebar-project-link--active': route.path === `/game/${game.id}`,
                }"
              >
                <Star class="h-[14px] w-[14px] shrink-0 text-yellow-500" />
                <span class="kosmos-sidebar-project-label">{{ game.name }}</span>
              </RouterLink>

              <div
                v-if="gamesStore.favorites.length > 5"
                class="px-3 py-1 text-xs text-muted-foreground"
              >
                +{{ gamesStore.favorites.length - 5 }} {{ t("sidebar.more") }}
              </div>
            </template>
          </div>

          <div class="kosmos-sidebar-footer">
            <RouterLink
              to="/settings"
              class="kosmos-sidebar-btn"
              :class="{ 'kosmos-sidebar-btn--active': route.path === '/settings' }"
            >
              <Settings class="h-[18px] w-[18px] shrink-0" />
              <span class="truncate">{{ t("sidebar.settings") }}</span>
            </RouterLink>
          </div>
        </template>
      </aside>
    </div>

    <div
      v-if="!mobile && !hidden"
      class="kosmos-sidebar-resize-handle"
      @mousedown="handleResizeStart"
    >
      <div class="kosmos-resize-handle-line" />
    </div>
  </div>
</template>
